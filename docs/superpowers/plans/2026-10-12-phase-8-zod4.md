- [ ] **Step 1: Write `e2e/specs/input-validation.spec.ts`**

Add one typed helper. Avoid `any`, because the lint config uses the type-checked presets:

```ts
type TrpcErrorData = {
  code: string;
  zodError: { formErrors: string[]; fieldErrors: Record<string, string[] | undefined> } | null;
};
async function trpcMutation(
  request: APIRequestContext, path: string, input: unknown,
): Promise<{ status: number; error?: TrpcErrorData }>
```
It POSTs `{ data: { json: input } }` (the superjson envelope) to `/api/trpc/${path}`, without batching. It returns `res.status()` and, on an error response, `body.error.json.data`. Tests pass `page.request`, not the standalone `request` fixture, so the admin session cookies from `ADMIN_AUTH` are sent (`changeState` is protected, `member.create` is admin-only).

Tests and their assertions:

```ts
test('event.create rejects an invalid email with a flattened zodError', async ({ page }) => {
  const { status, error } = await trpcMutation(page.request, 'event.create', {
    name: 'Zod Reject E2E', isPrivate: true, email: 'not-an-email',
  });
  expect(status).toBe(400);
  expect(error?.code).toBe('BAD_REQUEST');
  expect(error?.zodError?.formErrors).toEqual([]);
  expect(Object.keys(error?.zodError?.fieldErrors ?? {})).toEqual(['email']);
  expect(error?.zodError?.fieldErrors.email).toHaveLength(1);
  await expect(getEventByName('Zod Reject E2E')).rejects.toThrow();
});

test('event.changeState accepts EventState values and rejects anything else', async ({ page }) => {
  const event = await createTestEvent({ name: 'Zod State E2E' });
  for (const state of [7, 'CONFIRMED']) {
    const { status, error } = await trpcMutation(page.request, 'event.changeState', { id: event.id, state });
    expect(status).toBe(400);
    expect(Object.keys(error?.zodError?.fieldErrors ?? {})).toEqual(['state']);
  }
  const ok = await trpcMutation(page.request, 'event.changeState', { id: event.id, state: 2 });
  expect(ok.status).toBe(200);
  expect((await getEventByName('Zod State E2E')).state).toBe(2);
});

test('event.changeState rejects a malformed id', async ({ page }) => {
  const { status, error } = await trpcMutation(page.request, 'event.changeState', { id: 'not a cuid', state: 2 });
  expect(status).toBe(400);
  expect(Object.keys(error?.zodError?.fieldErrors ?? {})).toEqual(['id']);
});

test('member.create keeps the custom Czech role message', async ({ page }) => {
  const { status, error } = await trpcMutation(page.request, 'member.create', {
    email: 'zod-role@e2e.example.com', name: 'Zod Role', role: 'guest',
  });
  expect(status).toBe(400);
  expect(error?.zodError?.fieldErrors.role).toEqual(['Lze přiřadit pouze roli Člen nebo Admin.']);
});
```

If the Task 1 merge changed a procedure's input fields (for example the timezone fix reshaping `start`/`end`), keep each test's intent and adjust only the payload.
